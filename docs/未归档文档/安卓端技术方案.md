```kotlin
class MonitorAccessibilityService : AccessibilityService() {

    override fun onAccessibilityEvent(event: AccessibilityEvent) {
        when (event.eventType) {
            AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED -> {
                // 前台 App / Activity 切换
                handleWindowChange(event)
            }
            AccessibilityEvent.TYPE_NOTIFICATION_STATE_CHANGED -> {
                // 通知到达
                handleNotification(event)
            }
        }
    }

    override fun onInterrupt() {
        // 无障碍服务被系统中断
    }
}
```

### 3.3 前台 App 切换识别逻辑

`TYPE_WINDOW_STATE_CHANGED` 事件的 `packageName` 即为当前前台 App 的包名，`className` 为当前 Activity 的全名。

```kotlin
private var lastPackageName: String? = null
private var lastActivityClass: String? = null
private var lastSwitchTimeMs: Long = 0L

fun handleWindowChange(event: AccessibilityEvent) {
    val packageName = event.packageName?.toString() ?: return
    val className = event.className?.toString() ?: ""

    // 过滤系统界面（可选）
    if (isSystemUi(packageName)) return

    val now = System.currentTimeMillis()
    val duration = if (lastPackageName != null) now - lastSwitchTimeMs else 0L

    // 发出 app_foreground 事件
    EventCollector.emit(
        AppForegroundEvent(
            packageName = packageName,
            appName = getAppNameByPackage(packageName),
            activityName = className.substringAfterLast('.'),
            activityClass = className,
            fromPackage = lastPackageName,
            fromActivity = lastActivityClass,
            foregroundDurationMs = duration
        )
    )

    // 判断是否同一 App 内 Activity 切换
    if (packageName == lastPackageName && className != lastActivityClass) {
        EventCollector.emit(
            ActivityChangeEvent(
                packageName = packageName,
                appName = getAppNameByPackage(packageName),
                fromActivity = lastActivityClass,
                toActivity = className,
                // ...
            )
        )
    }

    lastPackageName = packageName
    lastActivityClass = className
    lastSwitchTimeMs = now
}
```

### 3.5 界面内容提取（App 内行为监控）

当检测到前台 Activity 变化时，通过 `AccessibilityNodeInfo` 遍历当前界面 UI 树，提取界面标题和内容信息，实现"你在微信里跟谁聊天"级别的监控。

```kotlin
// 节流：同一 App 同一屏幕 30 秒内不重复识别
private val screenContentCache = mutableMapOf<String, ScreenContentCache>()
private data class ScreenContentCache(
    val screenTitle: String, val timestampMs: Long
)

fun extractScreenContent(event: AccessibilityEvent) {
    val packageName = event.packageName?.toString() ?: return
    val now = System.currentTimeMillis()

    // 从窗口标题提取——很多 App 把当前页面名放在 event.text 中
    val windowTitle = event.text?.joinToString("")
        ?.takeIf { it.isNotBlank() }

    // 节流检查
    val cacheKey = "$packageName:${windowTitle}"
    val cached = screenContentCache[cacheKey]
    if (cached != null && now - cached.timestampMs < 30_000) return
    screenContentCache[cacheKey] = ScreenContentCache(windowTitle ?: "", now)

    // 获取根节点做深度提取（限 3 层深度，避免耗电）
    val rootNode = event.source
    val uiTexts = mutableListOf<String>()
    rootNode?.let { traverseNode(it, uiTexts, depth = 0, maxDepth = 3) }
    rootNode?.recycle()

    // 识别屏幕类型 + 生成事件
    val screenType = classifyScreen(packageName, event.className?.toString(), windowTitle)

    EventCollector.emit(
        ScreenContentEvent(
            packageName = packageName,
            appName = getAppNameByPackage(packageName),
            screenTitle = windowTitle ?: uiTexts.firstOrNull(),
            screenType = screenType,
            activityClass = event.className?.toString() ?: ""
        )
    )
}

// 遍历 UI 树，提取主要文本
fun traverseNode(node: AccessibilityNodeInfo, texts: MutableList<String>,
                 depth: Int, maxDepth: Int) {
    if (depth > maxDepth || texts.size >= 3) return

    node.text?.toString()?.takeIf { it.isNotBlank() }?.let {
        texts.add(it)
    }
    for (i in 0 until node.childCount) {
        node.getChild(i)?.let { traverseNode(it, texts, depth + 1, maxDepth) }
    }
}

// 屏幕类型分类
fun classifyScreen(pkg: String, cls: String?, title: String?): String {
    return when {
        pkg == "com.tencent.mm" && cls?.contains("ChattingUI") == true -> "chat"
        pkg == "com.tencent.mm" && cls?.contains("SnsTimeLineUI") == true -> "feed"
        pkg == "com.tencent.mm" && cls?.contains("BizTimeLineUI") == true -> "article"
        pkg == "com.tencent.mm" && cls?.contains("LauncherUI") == true -> "chat_list"
        cls?.contains("video", true) == true -> "video"
        cls?.contains("article", true) == true -> "article"
        cls?.contains("search", true) == true -> "search"
        cls?.contains("settings", true) == true -> "settings"
        else -> "unknown"
    }
}
```

## 5. 静默运行策略（核心设计）

本 App 的设计理念是**正常运行用户完全无感知，前台不留服务，通知栏永远不出现**。

### 5.1 零通知策略

Android 16 强制前台服务必须显示通知。为此，**放弃使用前台服务**，改用以下无通知的后台架构：

| 方案 | 说明 |
|------|------|
| **不启动 ForegroundService** | 完全避免前台服务通知，用户看不到任何图标或文字 |
| **仅依赖 AccessibilityService** | 无障碍服务本身具备系统级后台优先级，不需要前台服务保活 |
| **WorkManager 定时上报** | 事件上报使用 WorkManager 周期性任务，由系统调度，不需要前台服务 |
| **AlarmManager 保活检查** | 每 30 分钟用 AlarmManager 检查服务存活，不在前台则重新启动 |
| **BootReceiver 开机自启** | 设备解锁后自动启动 |

> ⚠️ **无前台服务的代价**：系统可能更频繁地杀掉进程，尤其在澎湃 OS 3 上。为此需要用户配合开启电池优化白名单和自启动管理。

### 5.2 故障通知机制

只有在以下情况才弹真实通知（使用普通 NotificationManager，不是前台服务通知）：

| 故障类型 | 检测方式 | 通知行为 |
|---------|---------|---------|
| 无障碍服务被系统关闭 | 定时检测 `isServiceRunning()` | 弹通知：「监控服务已停止，请重新开启」 |
| 网络中断 > 30 分钟 | 在 `EventReporter` 中累计离线时间 | 弹通知：「网络连接异常，数据将在恢复后上传」 |
| 事件队列积压 > 1000 条 | 检查 Room 中未上报事件数 | 弹通知：「事件积压过多，请检查网络或联系开发者」 |
| 系统低电量（< 15%） | 电量广播 | 不弹通知，自动降低上报频率 |

### 5.3 后台保活策略

| 策略 | 实现方式 | 重要性 |
|------|---------|--------|
| **AccessibilityService** | 系统对无障碍服务有保活优先级 | 核心依赖 |
| **AlarmManager 定时检测** | 每 30 分钟检查 `isServiceRunning()`，不在前台则重新启动 | 关键 |
| **电池优化白名单** | 引导用户将 App 设为"无限制" | 关键（必做） |
| **自启动管理** | 引导用户在安全中心 → 自启动管理 中开启 | 关键（必做） |
| **WorkManager 上传** | 系统调度的周期性任务，后台自动执行 | 辅助 |
| **BootReceiver** | 监听 `BOOT_COMPLETED`（设备解锁后延迟启动） | 基础 |
| **最近任务锁定** | 引导用户在最近任务列表中下拉锁定 App | 可选 |

```kotlin
class EventReporter(private val api: ApiService, private val db: AppDatabase) {

    // 定时批量上报
    fun uploadBatch() {
        val unsentEvents = db.eventDao().getUnsentEvents(limit = 50)
        if (unsentEvents.isEmpty()) return

        val batch = unsentEvents.map { it.toEvent() }
        val response = api.uploadEvents(EventsBatch(batch))

        if (response.isSuccessful) {
            db.eventDao().markUploaded(unsentEvents.map { it.eventId })
        } // 否则保留，下次重试
    }
}
```

### 8.7 App 崩溃/ANR 检测

无实时直接 API。使用以下间接方式：

```kotlin
// 方式一：通过前台 App 异常切换推断
// 当 App 切换到桌面且前台时长极短（< 3 秒）时，标记为可疑崩溃

// 方式二（API 30+）：查询 ApplicationExitInfo（不在实时事件流中）
val manager = getSystemService<ActivityManager>()!!
val reasons = manager.getHistoricalProcessExitReasons(packageName, 0, 0)
for (reason in reasons) {
    if (reason.reason == ActivityManager.PROCESS_STATE_CRASHED ||
        reason.reason == ActivityManager.PROCESS_STATE_DIED_ANR) {
        // 记录崩溃事件
    }
}
```

> ⚠️ `app_crashed` 事件**不是实时**的，通过定期检查（每 5 分钟）`ApplicationExitInfo` 获取最近崩溃记录。

```kotlin
registerReceiver(localeReceiver, filter, RECEIVER_EXPORTED)
```

无需额外权限。

## 9. 已知限制与风险

| 限制 | 说明 |
|------|------|
| **小米澎湃 OS 3 无障碍服务可能关闭** | HyperOS 可能自动关闭长时间运行的无障碍服务，需前台服务+定时检测恢复 |
| **无法监控锁屏界面 Enter 的密码** | 无障碍服务无法读取密码输入框内容 |
| **通知内容全部上报** | 根据用户配置，所有通知内容（含摘要）均上报。如需关闭需修改编译配置 |
| **电量消耗** | 无障碍服务 + 网络上报，目标日耗电 ≤ 5% |
| **Activity 切换的误报** | 某些 App 同一页面也会触发 `TYPE_WINDOW_STATE_CHANGED`（如 WebView 加载），需去重 |
| **多开/分身应用** | 分身空间中的 Activity 切换无障碍服务无法感知 |
| **系统桌面 Launcher** | 包名 `com.miui.home`，这是有效的 App 切换事件，不应过滤 |
| **无前台服务，系统可能杀进程** | 为保证通知栏完全无感，不启动 ForegroundService，系统（尤其澎湃 OS）可能更频繁地终止进程。AlarmManager 兜底重启 |
| **App 崩溃检测非实时** | `app_crashed` 事件通过定期检查 `ApplicationExitInfo` 获取，不是实时事件，可能有 5 分钟延迟 |
| **截屏检测范围** | `ScreenCaptureCallback` 只能检测本 App 在前台时的截屏。后台截屏需 `ContentObserver` 轮询，有延迟 |
| **通话状态仅上报元数据** | 不上报电话号码和通话内容，仅上报状态（来电/通话中/挂断） |

### 9.1 澎湃 OS 3 专项适配

| 问题 | 用户体验 | 方案 |
|------|---------|------|
| 无障碍服务自动关闭 | 监控中断 | 定时检测+通知引导重新开启 |
| 电池优化白名单 | App 被系统杀死 | App 启动时自动引导用户设置 |
| 自启动限制 | 重启后不自动启动 | 主动引导用户开启 |
| 通知通道被系统隐藏 | 前台服务通知无法显示 | 走系统预置通道 Fallback |
| 省电策略杀进程 | 事件丢失 | 事件持久化 Room，不丢失 |
